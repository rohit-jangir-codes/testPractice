 public Map<String, Double> getTotalAmount(List<classTransaction> transactions) {
 
 Set<String> seenTransectionId = new HashSet<>();
 Map<String, Double> result = new LinkedHashMap();
 
 for(classTransaction transaction: transactions) {
 
 // avoid duplictae
 if (!seenTransectionId.add(transaction.getTransectionId()) {
 continue;
 
 }
 
 String merchantId = transaction.getMerchantId();
 
 result.put(merchantId, result.getOrDefault(merchantId, 0.0)+transaction.getAmount());
 
 }
 
 return result;
